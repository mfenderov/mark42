- [ ] **Step 4: Replace the download callback with checksum-verified, separated extraction**

Find the existing download call and callback (from `download(url, tmpFile, (err) => {` to the closing `});`):

```js
download(url, tmpFile, (err) => {
  if (err) {
    try { fs.unlinkSync(tmpFile); } catch (_) {}
    console.error(`[mark42] Download failed: ${err.message}`);
    process.exit(1);
  }
  try {
    execFileSync('tar', ['xzf', tmpFile, '-C', nativeDir, 'mark42', 'mark42-server'], { stdio: 'inherit' });
    fs.chmodSync(path.join(nativeDir, 'mark42'), 0o755);
    fs.chmodSync(path.join(nativeDir, 'mark42-server'), 0o755);
    fs.unlinkSync(tmpFile);
    process.stderr.write(`[mark42] ✓ Done\n`);
  } catch (e) {
    console.error(`[mark42] Extraction failed: ${e.message}`);
    process.exit(1);
  }
});
```

Replace with:

```js
download(checksumsUrl, tmpChecksums, (checksumErr) => {
  if (checksumErr) {
    try { fs.unlinkSync(tmpChecksums); } catch (_) {}
    console.error(`[mark42] Failed to download checksums: ${checksumErr.message}`);
    process.exit(1);
  }

  download(url, tmpFile, (err) => {
    if (err) {
      try { fs.unlinkSync(tmpFile); } catch (_) {}
      try { fs.unlinkSync(tmpChecksums); } catch (_) {}
      console.error(`[mark42] Download failed: ${err.message}`);
      process.exit(1);
    }

    const checksumText = fs.readFileSync(tmpChecksums, 'utf8');
    try { fs.unlinkSync(tmpChecksums); } catch (_) {}

    const expectedLine = checksumText.split('\n').find(line => line.includes(tarball));
    if (!expectedLine) {
      try { fs.unlinkSync(tmpFile); } catch (_) {}
      console.error(`[mark42] Could not find checksum for ${tarball} in checksums.txt`);
      process.exit(1);
    }
    const expectedHash = expectedLine.trim().split(/\s+/)[0];

    const actualHash = crypto.createHash('sha256').update(fs.readFileSync(tmpFile)).digest('hex');
    if (actualHash !== expectedHash) {
      try { fs.unlinkSync(tmpFile); } catch (_) {}
      console.error(`[mark42] Checksum mismatch for ${tarball}!\n  expected: ${expectedHash}\n  actual:   ${actualHash}`);
      process.exit(1);
    }

    try {
      execFileSync('tar', ['xzf', tmpFile, '-C', nativeDir, 'mark42', 'mark42-server'], { stdio: 'inherit' });
    } catch (e) {
      try { fs.unlinkSync(tmpFile); } catch (_) {}
      console.error(`[mark42] Extraction failed: ${e.message}`);
      process.exit(1);
    }
    fs.unlinkSync(tmpFile);

    try {
      fs.chmodSync(path.join(nativeDir, 'mark42'), 0o755);
      fs.chmodSync(path.join(nativeDir, 'mark42-server'), 0o755);
    } catch (e) {
      console.error(`[mark42] Failed to set binary permissions: ${e.message}`);
      process.exit(1);
    }

    process.stderr.write(`[mark42] ✓ Done\n`);
  });
});
```
